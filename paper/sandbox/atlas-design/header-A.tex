\renewcommand{\baselinestretch}{1.30}

% Override tufte-book's \maketitle (uppercase macro chokes on pandoc subtitle)
\makeatletter
\renewcommand{\maketitle}{%
  \thispagestyle{empty}\null\vfill
  \begin{flushleft}
    {\Huge\bfseries\color{atlas}\@title\par}\vspace{1em}
    {\Large\itshape\@author\par}\vspace{0.5em}
    {\@date\par}
  \end{flushleft}\vfill\newpage}
\makeatother

% Neutralise tufte-book's automatic case-changing (chokes on pandoc \texorpdfstring etc.)
\makeatletter
\AtBeginDocument{%
  \let\MakeTextUppercase\@firstofone
  \let\MakeTextLowercase\@firstofone
  \DeclareRobustCommand{\smallcaps}[1]{\textsc{##1}}%
  \DeclareRobustCommand{\allcaps}[1]{##1}%
  \DeclareRobustCommand{\newlinetospace}[1]{##1}%
  \DeclareRobustCommand{\smallcapsspacing}[1]{##1}%
  \DeclareRobustCommand{\allcapsspacing}[1]{##1}%
}
\makeatother
