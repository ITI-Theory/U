% ---------------- Chapter opener
% Ghost number top-right, label + hairline + title bottom-left.
\makeatletter
\renewcommand{\@makechapterhead}[1]{%
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north east, inner sep=0pt]
      at ([xshift=-22mm, yshift=-22mm]current page.north east)
      {\fontsize{260}{260}\selectfont\color{atlassoft!30}\thechapter};
  \end{tikzpicture}%
  \tsticker%
  \vspace*{95mm}%
  {\noindent\sffamily\fontsize{8.5}{10}\selectfont\color{atlas}\MakeUppercase{Chapter \thechapter}\par}
  \vspace{3mm}%
  {\noindent\color{atlas}\rule{45mm}{0.6pt}\par}
  \vspace{8mm}%
  {\noindent\bfseries\color{atlasdeep}\fontsize{48}{54}\selectfont\raggedright #1\par}
  \vspace{18mm}%
}

  \thispagestyle{empty}%
  \tsticker%
  \vspace*{95mm}%
  {\noindent\bfseries\color{atlasdeep}\fontsize{42}{48}\selectfont\raggedright #1\par}
  \vspace{18mm}%
}

% ---------------- Part divider — full-page atlas-cyan slab + giant Roman numeral
\makeatletter
\renewcommand{\@endpart}{\thispagestyle{empty}\if@twoside\hbox{}\newpage\fi}
\renewcommand{\part}{\cleardoublepage\thispagestyle{empty}\@startpart}
\def\@startpart{\secdef\@part\@spart}
\def\@part[#1]#2{%
  \refstepcounter{part}%
  \addcontentsline{toc}{part}{\partname~\thepart\hspace{1em}#1}%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[atlas] (current page.north west) rectangle (current page.south east);
    \node[anchor=center, text=white] at (current page.center)
      {\fontsize{300}{300}\selectfont\bfseries\Roman{part}};
    \node[anchor=south, text=white, yshift=40mm] at (current page.center)
      {\sffamily\fontsize{10}{12}\selectfont\MakeUppercase{\partname~\Roman{part}}};
    \node[anchor=north, text=white, yshift=-50mm, text width=120mm, align=center] at (current page.center)
      {\bfseries\fontsize{28}{34}\selectfont #2};
  \end{tikzpicture}%
  \null\newpage\thispagestyle{empty}\null\newpage}
\makeatother
