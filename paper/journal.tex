% journal.tex — Professional journal typography for the SFT / USF papers
% Included via --include-in-header=journal.tex

% 1. Better typography
\usepackage{microtype}

% 2. Fonts (XeLaTeX fontspec)
\setmainfont{TeX Gyre Pagella}[Ligatures=TeX,Numbers=OldStyle]
\setsansfont{Lato}[Ligatures=TeX,BoldFont=Lato Bold]

% 3. Colours
\usepackage{xcolor}
\definecolor{heading}{RGB}{26,63,110}
\definecolor{accent}{RGB}{0,160,120}
\definecolor{rulecolor}{RGB}{180,190,200}

% Defer hyperref colour override
\makeatletter
\AtBeginDocument{\@ifpackageloaded{hyperref}{\hypersetup{linkcolor=heading,citecolor=heading,urlcolor=accent}}{}}
\makeatother

% 4. Section headings
\usepackage{titlesec}
\titleformat{\part}[display]{\LARGE\bfseries\sffamily\color{heading}\centering}{Part \thepart}{12pt}{\Huge\bfseries}
\titleformat{\section}{\Large\bfseries\sffamily\color{heading}}{\thesection}{0.8em}{}[\vspace{-4pt}{\color{rulecolor}\hrule height 0.5pt}\vspace{2pt}]
\titleformat{\subsection}{\large\bfseries\sffamily\color{heading}}{\thesubsection}{0.8em}{}
\titleformat{\subsubsection}{\normalsize\itshape\sffamily\color{heading}}{\thesubsubsection}{0.8em}{}
\titlespacing*{\section}{0pt}{14pt plus 2pt minus 2pt}{4pt}
\titlespacing*{\subsection}{0pt}{10pt plus 2pt minus 2pt}{2pt}
\titlespacing*{\subsubsection}{0pt}{8pt plus 2pt minus 2pt}{2pt}

% 5. Running headers
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE,RO]{\footnotesize\thepage}
\fancyhead[RE]{\footnotesize\sffamily\color{heading}\nouppercase{\leftmark}}
\fancyhead[LO]{\footnotesize\sffamily\color{heading}\nouppercase{\rightmark}}
\renewcommand{\headrulewidth}{0.4pt}
\makeatletter\renewcommand{\headrule}{{\color{rulecolor}\hrule width\headwidth height\headrulewidth}}\makeatother
\setlength{\headheight}{14pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\footnotesize\thepage}\renewcommand{\headrulewidth}{0pt}}

% 6. longtable fix (two-column mode only)
\makeatletter
\AtBeginDocument{\if@twocolumn\renewenvironment{longtable}[2][c]{\begin{table*}\centering\begin{tabular}{#2}}{\end{tabular}\end{table*}}\let\endhead\relax\let\endfirsthead\relax\let\endfoot\relax\let\endlastfoot\relax\fi}
\makeatother

% 7. Smaller code font
\makeatletter\AtBeginDocument{\@ifpackageloaded{fancyvrb}{\fvset{fontsize=\small}}{}}\makeatother
